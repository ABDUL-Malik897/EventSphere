import mongoose from "mongoose";
import { randomBytes } from "crypto";
import { z } from "zod";

import Event from "../models/Event.js";
import Booking from "../models/Booking.js";

const reservationSchema = z.object({
    eventId: z.string().regex(/^[a-f\d]{24}$/i),
    ticketTypeId: z.string().regex(/^[a-f\d]{24}$/i),
    quantity: z.coerce.number().int().min(1).max(10)
}).strict();

// CREATE A TICKET RESERVATION
export const createReservation = async (req, res) => {
    const validation = reservationSchema.safeParse(req.body);

    if (!validation.success) {
        return res.status(400).json({
            success: false,
            message: "Invalid booking details.",
            errors: validation.error.issues
        });
    }

    const { eventId, ticketTypeId, quantity } = validation.data;

    const eventObjectId = new mongoose.Types.ObjectId(eventId);
    const ticketObjectId = new mongoose.Types.ObjectId(ticketTypeId);

    const session = await mongoose.startSession();

    let createdBooking = null;

    try {
        await session.withTransaction(async () => {

            /*
             * Atomically check availability and increase
             * the sold counter for the selected ticket.
             *
             * The availability condition and inventory
             * update happen as one database operation.
             */

            const updatedEvent = await Event.findOneAndUpdate(
                {
                    _id: eventObjectId,
                    status: "published",
                    date: { $gt: new Date() },

                    $expr: {
                        $let: {
                            vars: {
                                selectedTicket: {
                                    $arrayElemAt: [
                                        {
                                            $filter: {
                                                input: {
                                                    $ifNull: [
                                                        "$ticketTypes",
                                                        []
                                                    ]
                                                },
                                                as: "ticket",
                                                cond: {
                                                    $eq: [
                                                        "$$ticket._id",
                                                        ticketObjectId
                                                    ]
                                                }
                                            }
                                        },
                                        0
                                    ]
                                }
                            },

                            in: {
                                $and: [
                                    {
                                        $ne: [
                                            "$$selectedTicket",
                                            null
                                        ]
                                    },
                                    {
                                        $lte: [
                                            {
                                                $add: [
                                                    {
                                                        $ifNull: [
                                                            "$$selectedTicket.sold",
                                                            0
                                                        ]
                                                    },
                                                    quantity
                                                ]
                                            },
                                            "$$selectedTicket.quantity"
                                        ]
                                    }
                                ]
                            }
                        }
                    }
                },

                [
                    {
                        $set: {
                            ticketTypes: {
                                $map: {
                                    input: "$ticketTypes",
                                    as: "ticket",

                                    in: {
                                        $cond: [
                                            {
                                                $eq: [
                                                    "$$ticket._id",
                                                    ticketObjectId
                                                ]
                                            },

                                            {
                                                $mergeObjects: [
                                                    "$$ticket",
                                                    {
                                                        sold: {
                                                            $add: [
                                                                {
                                                                    $ifNull: [
                                                                        "$$ticket.sold",
                                                                        0
                                                                    ]
                                                                },
                                                                quantity
                                                            ]
                                                        }
                                                    }
                                                ]
                                            },

                                            "$$ticket"
                                        ]
                                    }
                                }
                            }
                        }
                    }
                ],

                {
                    new: true,
                    session,
                    updatePipeline: true
                }
            );

            if (!updatedEvent) {
                const error = new Error(
                    "Tickets are unavailable, the event has ended, or the event is not published."
                );

                error.code = "TICKET_UNAVAILABLE";

                throw error;
            }

            const selectedTicket = updatedEvent.ticketTypes.find(
                (ticket) =>
                    ticket._id.toString() === ticketTypeId
            );

            if (!selectedTicket) {
                const error = new Error("Ticket category not found.");
                error.code = "TICKET_UNAVAILABLE";
                throw error;
            }

            const bookingReference =
                `ES-${randomBytes(5).toString("hex").toUpperCase()}`;

            const expiresAt = new Date(
                Date.now() + 10 * 60 * 1000
            );

            const totalAmount =
                selectedTicket.price * quantity;

            const [booking] = await Booking.create(
                [
                    {
                        bookingReference,

                        user: req.user._id,

                        event: updatedEvent._id,

                        ticketTypeId: selectedTicket._id,

                        eventTitle: updatedEvent.title,

                        eventDate: updatedEvent.date,

                        venue: updatedEvent.venue,

                        city: updatedEvent.city,

                        imageUrl: updatedEvent.imageUrl,

                        ticketTypeName: selectedTicket.name,

                        quantity,

                        unitPrice: selectedTicket.price,

                        totalAmount,

                        currency: updatedEvent.currency || "INR",

                        status: "PENDING_PAYMENT",

                        expiresAt
                    }
                ],
                { session }
            );

            createdBooking = booking;
        });

        res.status(201).json({
            success: true,
            message: "Tickets reserved successfully. Complete payment within 10 minutes.",
            booking: createdBooking
        });

    } catch (error) {
        console.error("Create reservation error:", error);

        if (error.code === "TICKET_UNAVAILABLE") {
            return res.status(409).json({
                success: false,
                message: error.message
            });
        }

        res.status(500).json({
            success: false,
            message: "Unable to reserve tickets."
        });

    } finally {
        await session.endSession();
    }
};

// CANCEL PENDING RESERVATION
export const cancelReservation = async (req, res) => {
    if (!mongoose.isValidObjectId(req.params.id)) {
        return res.status(400).json({
            success: false,
            message: "Invalid booking ID."
        });
    }

    const session = await mongoose.startSession();

    try {
        await session.withTransaction(async () => {

            const booking = await Booking.findOne({
                _id: req.params.id,
                user: req.user._id,
                status: "PENDING_PAYMENT"
            }).session(session);

            if (!booking) {
                const error = new Error(
                    "Pending booking not found or already processed."
                );

                error.code = "BOOKING_NOT_CANCELLABLE";

                throw error;
            }

            const inventoryResult = await Event.updateOne(
                {
                    _id: booking.event,

                    ticketTypes: {
                        $elemMatch: {
                            _id: booking.ticketTypeId,
                            sold: { $gte: booking.quantity }
                        }
                    }
                },

                {
                    $inc: {
                        "ticketTypes.$.sold": -booking.quantity
                    }
                },

                { session }
            );

            if (inventoryResult.modifiedCount !== 1) {
                throw new Error(
                    "Unable to release reserved ticket inventory."
                );
            }

            booking.status = "CANCELLED";

            await booking.save({ session });
        });

        res.status(200).json({
            success: true,
            message: "Reservation cancelled and tickets released."
        });

    } catch (error) {
        console.error("Cancel reservation error:", error);

        if (error.code === "BOOKING_NOT_CANCELLABLE") {
            return res.status(404).json({
                success: false,
                message: error.message
            });
        }

        res.status(500).json({
            success: false,
            message: "Unable to cancel reservation."
        });

    } finally {
        await session.endSession();
    }
};

import mongoose from "mongoose";
import { createHmac, timingSafeEqual } from "crypto";
import { z } from "zod";

import Booking from "../models/Booking.js";
import { getRazorpay } from "../config/razorpay.js";
import ProcessedWebhook from "../models/ProcessedWebhook.js";

const orderSchema = z.object({
    bookingId: z.string().regex(/^[a-f\d]{24}$/i)
}).strict();

const verifySchema = z.object({
    bookingId: z.string().regex(/^[a-f\d]{24}$/i),
    razorpay_payment_id: z.string().min(1),
    razorpay_order_id: z.string().min(1),
    razorpay_signature: z.string().min(1)
}).strict();

const toPaise = (amount) => Math.round(amount * 100);

// CREATE RAZORPAY ORDER
export const createPaymentOrder = async (req, res) => {
    try {
        const validation = orderSchema.safeParse(req.body);

        if (!validation.success) {
            return res.status(400).json({
                success: false,
                message: "Invalid booking ID."
            });
        }

        const booking = await Booking.findOne({
            _id: validation.data.bookingId,
            user: req.user._id
        });

        if (!booking) {
            return res.status(404).json({
                success: false,
                message: "Booking not found."
            });
        }

        if (booking.status !== "PENDING_PAYMENT") {
            return res.status(409).json({
                success: false,
                message: "This booking is no longer awaiting payment."
            });
        }

        const remainingTime =
            new Date(booking.expiresAt).getTime() - Date.now();

        if (remainingTime <= 0) {
            return res.status(409).json({
                success: false,
                message: "Your reservation has expired. Please book again."
            });
        }

        const razorpay = getRazorpay();

        const amount = toPaise(booking.totalAmount);

        if (!Number.isSafeInteger(amount) || amount < 100) {
            return res.status(400).json({
                success: false,
                message: "Invalid payment amount."
            });
        }

        let order;

        // Reuse an order already created for this booking.
        if (booking.razorpayOrderId) {
            order = await razorpay.orders.fetch(
                booking.razorpayOrderId
            );

            if (
                order.amount !== amount ||
                order.currency !== booking.currency
            ) {
                return res.status(409).json({
                    success: false,
                    message: "Payment order does not match this booking."
                });
            }

            if (order.status === "paid") {
                return res.status(409).json({
                    success: false,
                    message: "This payment order has already been paid. Refresh your bookings."
                });
            }

        } else {
            order = await razorpay.orders.create({
                amount,
                currency: booking.currency,
                receipt: booking.bookingReference,
                notes: {
                    bookingId: booking._id.toString(),
                    bookingReference: booking.bookingReference
                }
            });

            booking.razorpayOrderId = order.id;
            booking.paymentStatus = "ORDER_CREATED";

            await booking.save();
        }

        res.status(201).json({
            success: true,
            keyId: process.env.RAZORPAY_KEY_ID,
            orderId: order.id,
            amount: order.amount,
            currency: order.currency,
            bookingId: booking._id,
            checkoutTimeoutSeconds: Math.max(
                1,
                Math.floor(remainingTime / 1000)
            ),
            prefill: {
                name: req.user.name,
                email: req.user.email
            }
        });

    } catch (error) {
        console.error("Create payment order error:", error);

        res.status(500).json({
            success: false,
            message: "Unable to initialize payment."
        });
    }
};

// VERIFY PAYMENT
export const verifyPayment = async (req, res) => {
    try {
        const validation = verifySchema.safeParse(req.body);

        if (!validation.success) {
            return res.status(400).json({
                success: false,
                message: "Invalid payment verification details."
            });
        }

        const {
            bookingId,
            razorpay_payment_id,
            razorpay_order_id,
            razorpay_signature
        } = validation.data;

        const booking = await Booking.findOne({
            _id: bookingId,
            user: req.user._id
        });

        if (!booking || !booking.razorpayOrderId) {
            return res.status(404).json({
                success: false,
                message: "Payment booking not found."
            });
        }

        // Never trust an order ID just because the browser supplied it.
        const savedOrderId = booking.razorpayOrderId;

        if (savedOrderId !== razorpay_order_id) {
            return res.status(400).json({
                success: false,
                message: "Payment order mismatch."
            });
        }

        if (!process.env.RAZORPAY_KEY_SECRET) {
            throw new Error("Razorpay Key Secret is missing.");
        }

        // Verify using the order ID stored in MongoDB.
        const expectedSignature = createHmac(
            "sha256",
            process.env.RAZORPAY_KEY_SECRET
        )
            .update(`${savedOrderId}|${razorpay_payment_id}`)
            .digest("hex");

        const expectedBuffer = Buffer.from(expectedSignature, "hex");
        const suppliedBuffer = Buffer.from(razorpay_signature, "hex");

        const signatureValid =
            expectedBuffer.length === suppliedBuffer.length &&
            timingSafeEqual(expectedBuffer, suppliedBuffer);

        if (!signatureValid) {
            return res.status(400).json({
                success: false,
                message: "Payment signature verification failed."
            });
        }

        const razorpay = getRazorpay();

        // Fetch the actual provider-side payment.
        let payment = await razorpay.payments.fetch(
            razorpay_payment_id
        );

        const expectedAmount = toPaise(booking.totalAmount);

        if (
            payment.order_id !== savedOrderId ||
            payment.amount !== expectedAmount ||
            payment.currency !== booking.currency
        ) {
            return res.status(400).json({
                success: false,
                message: "Payment amount, currency or order does not match."
            });
        }

        // Capture an authorized payment before confirming the booking.
        if (payment.status === "authorized") {
            payment = await razorpay.payments.capture(
                payment.id,
                expectedAmount,
                booking.currency
            );
        }

        // Do not issue tickets for authorized or failed payments.
        if (payment.status !== "captured" || !payment.captured) {
            return res.status(409).json({
                success: false,
                message: "Payment has not been captured yet. Your booking is not confirmed.",
                paymentStatus: payment.status
            });
        }

        // Idempotency: a repeated valid callback must not create
        // another confirmation.
        if (booking.status === "CONFIRMED") {
            if (booking.razorpayPaymentId === payment.id) {
                return res.status(200).json({
                    success: true,
                    message: "Payment was already verified.",
                    booking
                });
            }

            return res.status(409).json({
                success: false,
                message: "This booking has already been confirmed with another payment."
            });
        }

        if (booking.status !== "PENDING_PAYMENT") {
            return res.status(409).json({
                success: false,
                message: "Reservation is no longer active. Contact support if payment was captured."
            });
        }

        if (new Date(booking.expiresAt) <= new Date()) {
            return res.status(409).json({
                success: false,
                message: "Reservation expired. Contact support if payment was captured."
            });
        }

        const confirmedBooking = await Booking.findOneAndUpdate(
            {
                _id: booking._id,
                user: req.user._id,
                status: "PENDING_PAYMENT",
                expiresAt: { $gt: new Date() },
                razorpayOrderId: savedOrderId
            },
            {
                $set: {
                    status: "CONFIRMED",
                    paymentStatus: "CAPTURED",
                    razorpayPaymentId: payment.id,
                    paidAt: new Date()
                }
            },
            {
                new: true
            }
        );

        if (!confirmedBooking) {
            return res.status(409).json({
                success: false,
                message: "Booking status changed while payment was being verified. Contact support if payment was captured."
            });
        }

        res.status(200).json({
            success: true,
            message: "Payment verified! Your booking is confirmed.",
            booking: confirmedBooking
        });

    } catch (error) {
        console.error("Payment verification error:", error);

        res.status(500).json({
            success: false,
            message: "Unable to verify payment. Please check your booking status before retrying."
        });
    }
};

// RAZORPAY WEBHOOK
export const razorpayWebhook = async (req, res) => {
    let eventId;

    try {
        if (!Buffer.isBuffer(req.body)) {
            return res.status(415).json({
                success: false,
                message: "Raw webhook body required."
            });
        }

        const webhookSecret = process.env.RAZORPAY_WEBHOOK_SECRET;

        if (!webhookSecret) {
            console.error("Razorpay webhook secret is missing.");

            return res.status(503).json({
                success: false,
                message: "Webhook is not configured."
            });
        }

        const signature = req.get("X-Razorpay-Signature");

        if (!signature || !/^[a-f0-9]{64}$/i.test(signature)) {
            return res.status(400).json({
                success: false,
                message: "Invalid webhook signature."
            });
        }

        const expected = createHmac(
            "sha256",
            webhookSecret
        )
            .update(req.body)
            .digest();

        const received = Buffer.from(signature, "hex");

        if (
            expected.length !== received.length ||
            !timingSafeEqual(expected, received)
        ) {
            return res.status(400).json({
                success: false,
                message: "Webhook signature verification failed."
            });
        }

        const payload = JSON.parse(req.body.toString("utf8"));

        eventId = req.get("x-razorpay-event-id");

        if (!eventId || !payload.event) {
            return res.status(400).json({
                success: false,
                message: "Webhook event information is missing."
            });
        }

        // Ignore duplicate deliveries.
        try {
            await ProcessedWebhook.create({
                eventId,
                eventType: payload.event
            });
        } catch (error) {
            if (error.code === 11000) {
                return res.status(200).json({
                    success: true,
                    message: "Webhook already processed."
                });
            }

            throw error;
        }

        // We only fulfil bookings on an order.paid event.
        if (payload.event !== "order.paid") {
            return res.status(200).json({
                success: true,
                message: "Webhook acknowledged."
            });
        }

        const order = payload.payload?.order?.entity;
        const payment = payload.payload?.payment?.entity;

        if (!order?.id || !payment?.id) {
            throw new Error("Payment or order information missing from webhook.");
        }

        const booking = await Booking.findOne({
            razorpayOrderId: order.id
        });

        if (!booking) {
            console.warn("Webhook order has no matching booking:", order.id);

            return res.status(200).json({
                success: true,
                message: "No matching booking."
            });
        }

        const expectedAmount = toPaise(booking.totalAmount);

        if (
            payment.order_id !== booking.razorpayOrderId ||
            payment.amount !== expectedAmount ||
            payment.currency !== booking.currency ||
            payment.status !== "captured" ||
            payment.captured !== true
        ) {
            throw new Error("Webhook payment details do not match the booking.");
        }

        if (
            booking.status === "CONFIRMED" &&
            booking.razorpayPaymentId === payment.id
        ) {
            return res.status(200).json({
                success: true,
                message: "Booking was already confirmed."
            });
        }

        if (
            booking.status === "PENDING_PAYMENT" &&
            new Date(booking.expiresAt) > new Date()
        ) {
            await Booking.findOneAndUpdate(
                {
                    _id: booking._id,
                    status: "PENDING_PAYMENT",
                    expiresAt: { $gt: new Date() },
                    razorpayOrderId: order.id
                },
                {
                    $set: {
                        status: "CONFIRMED",
                        paymentStatus: "CAPTURED",
                        razorpayPaymentId: payment.id,
                        paidAt: new Date()
                    }
                }
            );
        } else {
            // Do not restore tickets automatically after their
            // reservation has expired or been cancelled.
            console.error(
                "Captured payment needs reconciliation:",
                booking.bookingReference,
                payment.id
            );
        }

        return res.status(200).json({
            success: true,
            message: "Webhook processed."
        });

    } catch (error) {
        console.error("Razorpay webhook error:", error);

        // Remove the event record so Razorpay can retry.
        if (eventId) {
            await ProcessedWebhook.deleteOne({ eventId }).catch(
                () => {}
            );
        }

        return res.status(500).json({
            success: false,
            message: "Webhook processing failed."
        });
    }
};

import mongoose from "mongoose";
import { z } from "zod";

import Cart from "../models/Cart.js";

const addItemSchema = z.object({

const getOrCreateCart = async (userId) => {
    let cart = await Cart.findOne({ user: userId });

    if (!cart) {
        try {
            cart = await Cart.create({
                user: userId,
                items: []
            });
        } catch (error) {
            // Another request may have created this user's cart first.
            if (error.code !== 11000) throw error;
            cart = await Cart.findOne({ user: userId });
        }
    }

    return cart;
};

const findAvailableTicket = async (eventId, ticketTypeId) => {
    if (
        !mongoose.isValidObjectId(eventId) ||
        !mongoose.isValidObjectId(ticketTypeId)
    ) {
        return { error: "Invalid event or ticket type." };
    }

    const event = await Event.findOne({
        _id: eventId,
        status: "published",
        date: { $gt: new Date() }
    });

    if (!event) {
        return { error: "This event is unavailable or has already started." };
    }

    const ticket = event.ticketTypes.id(ticketTypeId);

    if (!ticket) {
        return { error: "This ticket category no longer exists." };
    }

    const available = Math.max(0, ticket.quantity - ticket.sold);

    if (available < 1) {
        return { error: "This ticket category is sold out." };
    }

    return { event, ticket, available };
};

// GET MY CART
export const getMyCart = async (req, res) => {
    try {
        const cart = await Cart.findOne({ user: req.user._id })
            .populate({
                path: "items.event",
                select: "title slug date time venue city imageUrl currency status ticketTypes"
            });

        if (!cart) {
            return res.status(200).json({
                success: true,
                cart: { items: [] },
                ticketCount: 0
            });
        }

        const items = cart.items.map((item) => {
            const event = item.event;

            const ticket = event?.ticketTypes?.id(item.ticketTypeId);

            const availableQuantity = ticket
                ? Math.max(0, ticket.quantity - ticket.sold)
                : 0;

            const unavailable =
                !event ||
                event.status !== "published" ||
                new Date(event.date) <= new Date() ||
                !ticket ||
                availableQuantity < item.quantity;

            return {
                _id: item._id,
                event: event ? {
                    _id: event._id,
                    title: event.title,
                    slug: event.slug,
                    date: event.date,
                    time: event.time,
                    venue: event.venue,
                    city: event.city,
                    imageUrl: event.imageUrl,
                    currency: event.currency
                } : null,
                ticketType: ticket ? {
                    _id: ticket._id,
                    name: ticket.name,
                    description: ticket.description,
                    price: ticket.price
                } : null,
                quantity: item.quantity,
                availableQuantity,
                unavailable
            };
        });

        const ticketCount = items.reduce(
            (total, item) => total + item.quantity,
            0
        );

        return res.status(200).json({
            success: true,
            cart: {
                _id: cart._id,
                items
            },
            ticketCount
        });
    } catch (error) {

// ADD ITEM OR INCREASE EXISTING QUANTITY
export const addCartItem = async (req, res) => {
    try {
        const result = addItemSchema.safeParse(req.body);

        if (!result.success) {
            return res.status(400).json({
                success: false,
                message: "Invalid cart item details.",
                errors: result.error.issues
            });
        }

        const { eventId, ticketTypeId, quantity } = result.data;

        const availability = await findAvailableTicket(
            eventId,
            ticketTypeId
        );

        if (availability.error) {
            return res.status(400).json({
                success: false,
                message: availability.error
            });
        }

        const cart = await getOrCreateCart(req.user._id);

        const existingItem = cart.items.find((item) =>
            item.event.toString() === eventId &&
            item.ticketTypeId.toString() === ticketTypeId
        );

        const newQuantity = existingItem
            ? existingItem.quantity + quantity
            : quantity;

        if (newQuantity > availability.available) {
            return res.status(400).json({
                success: false,
                message: `Only ${availability.available} ticket(s) are currently available.`
            });
        }

        if (existingItem) {
            existingItem.quantity = newQuantity;
        } else {
            cart.items.push({
                event: eventId,
                ticketTypeId,
                quantity
            });
        }

        await cart.save();

        return res.status(200).json({
            success: true,
            message: "Ticket added to your cart.",
            ticketCount: cart.items.reduce(
                (total, item) => total + item.quantity,
                0
            )
        });
    } catch (error) {
        console.error("Add cart item error:", error);

        return res.status(500).json({
            success: false,
            message: "Could not add this ticket to your cart."
        });
    }
};

import mongoose from "mongoose";

import Booking from "../models/Booking.js";
import Event from "../models/Event.js";

let expiryTimer = null;
let isRunning = false;

export const expirePendingBookings = async () => {
    if (isRunning) return;

    isRunning = true;

    try {
        const expiredBookings = await Booking.find({
            status: "PENDING_PAYMENT",
            expiresAt: { $lte: new Date() }
        })
            .select("_id")
            .limit(100)
            .lean();

        for (const candidate of expiredBookings) {
            const session = await mongoose.startSession();

            try {
                await session.withTransaction(async () => {

                    // Claim the expired booking only once.
                    const booking = await Booking.findOneAndUpdate(
                        {
                            _id: candidate._id,
                            status: "PENDING_PAYMENT",
                            expiresAt: { $lte: new Date() }
                        },
                        {
                            $set: {
                                status: "EXPIRED"
                            }
                        },
                        {
                            new: true,
                            session
                        }
                    );

                    if (!booking) return;

                    // Release the inventory.
                    const result = await Event.updateOne(
                        {
                            _id: booking.event,
                            ticketTypes: {
                                $elemMatch: {
                                    _id: booking.ticketTypeId,
                                    sold: { $gte: booking.quantity }
                                }
                            }
                        },
                        {
                            $inc: {
                                "ticketTypes.$.sold": -booking.quantity
                            }
                        },
                        { session }
                    );

                    if (result.modifiedCount !== 1) {
                        throw new Error(
                            `Unable to release inventory for booking ${booking.bookingReference}`
                        );
                    }

                    console.log(
                        `Reservation expired: ${booking.bookingReference}`
                    );
                });

            } catch (error) {
                console.error(
                    `Expiry processing failed for ${candidate._id}:`,
                    error.message
                );
            } finally {
                await session.endSession();
            }
        }

    } catch (error) {
        console.error("Booking expiry job failed:", error.message);

    } finally {
        isRunning = false;
    }
};

export const startBookingExpiryJob = () => {
    if (expiryTimer) return;

    console.log("Booking expiry worker started.");

    // Check immediately on startup.
    expirePendingBookings();

    // Check every 60 seconds.
    expiryTimer = setInterval(() => {
        expirePendingBookings();
    }, 60 * 1000);
};

import {
    createContext,
    useCallback,
    useContext,
    useEffect,
    useMemo,
    useState
} from "react";

import api from "../services/api";
import { useAuth } from "./AuthContext";

const CartContext = createContext(null);

export const CartProvider = ({ children }) => {
    const { user, loading: authLoading } = useAuth();

    const canUseCart = ["ATTENDEE", "ORGANIZER"].includes(user?.role);

    const [items, setItems] = useState([]);
    const [ticketCount, setTicketCount] = useState(0);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState("");

    const refreshCart = useCallback(async () => {
        if (!canUseCart) {
            setItems([]);
            setTicketCount(0);
            setLoading(false);
            return null;
        }

        setLoading(true);

        try {
            const response = await api.get("/cart");

            const cartItems = response.data.cart?.items || [];

            setItems(cartItems);
            setTicketCount(
                response.data.ticketCount ??
                cartItems.reduce(
                    (total, item) => total + item.quantity,
                    0
                )
            );

            setError("");

            return response.data;
        } catch (err) {
            setError(
                err.response?.data?.message ||
                "Unable to load your cart."
            );

            throw err;
        } finally {
            setLoading(false);
        }
    }, [canUseCart]);

    useEffect(() => {
        if (authLoading) return;

        if (canUseCart) {
            refreshCart().catch(() => {});
        } else {
            setItems([]);
            setTicketCount(0);
            setLoading(false);
            setError("");
        }
    }, [authLoading, canUseCart, refreshCart]);

    const addItem = useCallback(async (item) => {
        const response = await api.post("/cart/items", item);

        await refreshCart();

        return response.data;
    }, [refreshCart]);

    const updateItem = useCallback(async (itemId, quantity) => {
        const response = await api.patch(
            `/cart/items/${itemId}`,
            { quantity }
        );

        await refreshCart();

        return response.data;
    }, [refreshCart]);

    const removeItem = useCallback(async (itemId) => {
        const response = await api.delete(
            `/cart/items/${itemId}`
        );

        await refreshCart();

        return response.data;
    }, [refreshCart]);

    const clearCart = useCallback(async () => {
        const response = await api.delete("/cart");

        await refreshCart();

        return response.data;
    }, [refreshCart]);

    const value = useMemo(() => ({
        items,
        ticketCount,
        loading,
        error,
        refreshCart,
        addItem,
        updateItem,
        removeItem,
        clearCart
    }), [
        items,
        ticketCount,
        loading,
        error,
        refreshCart,
        addItem,
        updateItem,
        removeItem,
        clearCart
    ]);

    return (
        <CartContext.Provider value={value}>
            {children}
        </CartContext.Provider>
    );
};

export const useCart = () => {
    const context = useContext(CartContext);

    if (!context) {
        throw new Error(
            "useCart must be used inside CartProvider"
        );
    }

    return context;
};

import api from "./api.js";

let checkoutScriptPromise = null;

// Load Razorpay Checkout JavaScript only once.
const loadRazorpayScript = () => {
    if (window.Razorpay) {
        return Promise.resolve(true);
    }

    if (checkoutScriptPromise) {
        return checkoutScriptPromise;
    }

    checkoutScriptPromise = new Promise((resolve) => {
        const script = document.createElement("script");

        script.src = "https://checkout.razorpay.com/v1/checkout.js";
        script.async = true;

        script.onload = () => {
            resolve(Boolean(window.Razorpay));
        };

        script.onerror = () => {
            checkoutScriptPromise = null;
            resolve(false);
        };

        document.body.appendChild(script);
    });

    return checkoutScriptPromise;
};

// Create an order and open Razorpay Checkout.
export const startRazorpayCheckout = async (bookingId) => {
    const scriptLoaded = await loadRazorpayScript();

    if (!scriptLoaded) {
        throw new Error(
            "Unable to load Razorpay Checkout. Check your internet connection."
        );
    }

    // Ask our backend to create or retrieve the payment order.
    const orderResponse = await api.post(
        "/payments/create-order",
        { bookingId }
    );

    const order = orderResponse.data;

    return new Promise((resolve) => {
        let completed = false;

        // Prevent multiple callbacks from resolving the same checkout.
        const finish = (result) => {
            if (completed) return;

            completed = true;
            resolve(result);
        };

        const checkout = new window.Razorpay({
            key: order.keyId,

            amount: order.amount,

            currency: order.currency,

            name: "EventSphere",

            description: "Event ticket reservation",

            order_id: order.orderId,

            prefill: {
                name: order.prefill.name,
                email: order.prefill.email
            },

            theme: {
                color: "#f15b3e"
            },

            modal: {
                confirm_close: true,

                ondismiss: () => {
                    finish({
                        status: "dismissed",
                        message: "Payment checkout was closed."
                    });
                }
            },

            // This executes after a successful Checkout payment.
            handler: async (paymentResponse) => {
                try {
                    const verificationResponse = await api.post(
                        "/payments/verify",
                        {
                            bookingId,

                            razorpay_payment_id:
                                paymentResponse.razorpay_payment_id,

                            razorpay_order_id:
                                paymentResponse.razorpay_order_id,

                            razorpay_signature:
                                paymentResponse.razorpay_signature
                        }
                    );

                    const confirmedBooking =
                        verificationResponse.data.booking;

                    if (
                        !verificationResponse.data.success ||
                        confirmedBooking?.status !== "CONFIRMED"
                    ) {
                        finish({
                            status: "pending",
                            message:
                                "Payment was received, but booking confirmation is still pending."
                        });

                        return;
                    }

                    finish({
                        status: "confirmed",
                        booking: confirmedBooking
                    });

                } catch (error) {
                    console.error(
                        "Payment verification failed:",
                        error
                    );

                    finish({
                        status: "verification_failed",
                        message:
                            error.response?.data?.message ||
                            "Payment could not be verified. Please check My Bookings before retrying."
                    });
                }
            }
        });

        // Listen for failed payment attempts.
        checkout.on("payment.failed", (response) => {
            finish({
                status: "failed",
                message:
                    response.error?.description ||
                    "Payment failed. You can try again."
            });
        });

        checkout.open();
    });
};
